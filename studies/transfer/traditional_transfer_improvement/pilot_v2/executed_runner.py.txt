"""Frozen FULL-data 25g ROW validation-only traditional transfer pilot v2."""
from __future__ import annotations
import argparse
import hashlib
import json
import sys
from pathlib import Path
import time
import numpy as np
import pandas as pd
import torch

ROOT = Path(__file__).resolve().parents[2]
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))
from src.qgeognn_al.resources import SOURCE_GRAPH_CACHE
from src.qgeognn_al.data import build_model_data
from src.qgeognn_al.models import load_predictor_checkpoint
from src.qgeognn_al.transfer import adaptation as a
from scripts.studies.run_filtered_full_data_benchmark import _read_authorized_truth

STUDY = ROOT / 'studies/transfer/traditional_transfer_improvement'
FROZEN = ROOT / 'studies/transfer/filtered_full_data_benchmark'
SOURCE_CHECKPOINT_SEED = 42
SOURCE = ROOT / 'studies/predictor/final_4g_qualification/runtime/row/seed_42/best.pt'
TARGET_CACHE = ROOT / 'studies/transfer/cross_column/data_audit/graph_cache_25g_only.pt'
METHODS = {'P0': ('current', 'adam'), 'P1': ('current', 'adam'),
           'P2': ('source_stats', 'adam'), 'P3': ('source_stats', 'lbfgs')}
FEATURES = ['sample_id', 'canonical_smiles', 'PE/EA', 'loading solvent', 'Density g/ml',
            'V/ul', 'Volume of loading solvent/ul']

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(',', ':')).encode()).hexdigest()

def write_json(path, value):
    path.write_text(json.dumps(value, indent=2, allow_nan=False) + '\n')

def prepare_context():
    frozen = json.loads((FROZEN / 'protocol.json').read_text())
    target_outer_seed = int(frozen['outer_seeds'][0])
    assert sha(FROZEN / 'filtered_canonical_25g.csv') == frozen['filtered_canonical_sha256']['25g']
    assert sha(SOURCE) == frozen['paper_style_current_v2']['source_checkpoint_sha256']
    schedule = pd.read_csv(FROZEN / 'split_manifest.csv')
    context = schedule.loc[schedule.column.eq('25g') & schedule.protocol.eq('row') & schedule.outer_seed.eq(target_outer_seed),
                           ['column', 'protocol', 'outer_seed', 'sample_id', 'role']].reset_index(drop=True)
    # Projected features exclude target endpoints and raw/verified retention times.
    feature = pd.read_csv(FROZEN / 'filtered_features_25g.csv', usecols=FEATURES)
    expected = pd.read_csv(FROZEN / 'filtered_canonical_25g.csv', usecols=['sample_id']).sample_id.tolist()
    assert feature.sample_id.is_unique and context.sample_id.is_unique
    assert set(context.role) == {'gradient_train', 'validation', 'test'}
    assert set(expected) == set(feature.sample_id) == set(context.sample_id)
    cache = dict(torch.load(SOURCE_GRAPH_CACHE, weights_only=False))
    cache.update(torch.load(TARGET_CACHE, weights_only=False))
    actual = feature.loc[feature.canonical_smiles.isin(cache), 'sample_id'].tolist()
    coverage = dict(expected_filtered_rows=len(expected), actual_graph_rows=len(actual),
                    missing_sample_ids=sorted(set(expected)-set(actual)), extra_sample_ids=sorted(set(actual)-set(expected)))
    if coverage['missing_sample_ids'] or coverage['extra_sample_ids']:
        raise RuntimeError(json.dumps(coverage))
    roles = {r: context.loc[context.role.eq(r), 'sample_id'].tolist() for r in ['gradient_train','validation','test']}
    assert not (set(roles['gradient_train']) & set(roles['validation']) or set(roles['gradient_train']) & set(roles['test']) or set(roles['validation']) & set(roles['test']))
    source_preprocessing = torch.load(SOURCE, weights_only=False, map_location='cpu')['preprocessing']
    # Fit API receives only train/validation graphs; no test graph or truth exists there.
    fit_ids = roles['gradient_train'] + roles['validation']
    fit_frame = feature.set_index('sample_id').loc[fit_ids].reset_index()
    truth = _read_authorized_truth(FROZEN / 'filtered_canonical_25g.csv', fit_ids)
    fit_frame[['V1_ml','V2_ml']] = truth
    atoms, angles = build_model_data(fit_frame, cache, None, source_preprocessing['scaler'])
    train = list(range(len(roles['gradient_train'])))
    valid = list(range(len(train), len(fit_ids)))
    preprocessing = dict(source_preprocessing)
    preprocessing['target_scales'] = a.fit_target_scales(atoms, train)
    preprocessing['fit_role'] = 'gradient_train'
    audit = dict(coverage, column='25g', protocol='row', target_outer_seed=target_outer_seed,
                 source_checkpoint_seed=SOURCE_CHECKPOINT_SEED, source_checkpoint_sha256=sha(SOURCE),
                 source_preprocessing_hash=digest(source_preprocessing), scaler=source_preprocessing['scaler'],
                 scaler_hash=digest(preprocessing['scaler']), target_scales=preprocessing['target_scales'],
                 split_counts={r:len(ids) for r,ids in roles.items()}, test_truth_used=False,
                 fit_sample_ids=fit_ids, test_sample_ids=roles['test'])
    assert audit['scaler_hash'] == digest(source_preprocessing['scaler'])
    return context, atoms, angles, train, valid, preprocessing, audit

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--run', action='store_true')
    parser.add_argument('--dry-run', action='store_true')
    parser.add_argument('--output', type=Path, default=STUDY / 'pilot_v2')
    args = parser.parse_args()
    if args.run and args.dry_run:
        parser.error('choose --run or --dry-run')
    out = args.output
    if (out / 'pilot_artifact_manifest.json').exists():
        raise RuntimeError('output already exists; preserve previous pilot artifacts')
    out.mkdir(parents=True, exist_ok=True)
    context, atoms, angles, train, valid, preprocessing, audit = prepare_context()
    context.to_csv(out / 'pilot_split_manifest.csv', index=False)
    protocol = dict(audit, status='PREFLIGHT_VERIFIED', head_initialization='source head copy',
                    shallow_scope='SourceAnchoredTransfer shallow: backbone.convs.4 + condition_branch + head',
                    adam_budget={'maximum_epochs':40, 'patience':15}, lbfgs_budget={'maximum_epochs':20, 'patience':10, 'max_iter':5},
                    loss='raw quantile target loss; no normalized-loss comparison',
                    post_step_train_loss_semantics='post-update eval-mode full train loss',
                    selection='validation combined NRMSE; train-derived endpoint standard deviations',
                    methods=METHODS)
    write_json(out / 'PILOT_PROTOCOL.json', protocol)
    if not args.run:
        print(json.dumps(audit, indent=2)); return
    torch.set_num_threads(1)
    results, history, inventories, drifts, metrics = {}, [], [], [], []
    started = time.time()
    for method, (bn, optimizer) in METHODS.items():
        model = load_predictor_checkpoint(SOURCE)
        a.configure_trainable(model, 'historical_shallow')
        for name, p in model.named_parameters():
            inventories.append(dict(method=method, stage='direct' if method=='P0' else 'B', parameter=name, numel=p.numel(), trainable=p.requires_grad))
        bn_before = a.snapshot_bn_buffers(model)
        affine_names = {f'{n}.{p}' for n,m in model.named_modules() if isinstance(m, torch.nn.modules.batchnorm._BatchNorm) for p,_ in m.named_parameters(recurse=False)}
        affine_before = {n:p.detach().clone() for n,p in model.named_parameters() if n in affine_names}
        config = dict(maximum_epochs=40 if optimizer=='adam' else 20, patience=15 if optimizer=='adam' else 10,
                      batch_size=2048, bn_policy=bn, optimizer=optimizer, learning_rate=1e-4 if optimizer=='adam' else 0.1,
                      weight_decay=1e-5 if optimizer=='adam' else 0, lbfgs_max_iter=5)
        if method=='P0':
            stages = {'B': a.train_target_adaptation(model, atoms, angles, train, valid, preprocessing,
                        mode='historical_shallow', seed=audit['target_outer_seed'], config=config)}
        else:
            fit = a.train_staged_target_adaptation(model, atoms, angles, train, valid, preprocessing,
                        stage_b_mode='historical_shallow', seed=audit['target_outer_seed'], stage_a_config=config, stage_b_config=config)
            stages = {'A':fit.stage_a, 'B':fit.stage_b}
        results[method] = {s:f.as_dict() for s,f in stages.items()}
        for stage, fit in stages.items():
            history.extend(dict(method=method, stage=stage, **row) for row in fit.history)
            metrics.append(dict(method=method, stage=stage, combined_nrmse=fit.validation_score, best_epoch=fit.best_epoch, epochs_run=fit.epochs_run, trainable_parameters=fit.trainable_parameters))
        truth, pred, _ = a.predict_point(model, atoms, angles, valid)
        assert np.isfinite(pred).all()
        drift = a.bn_buffer_drift(model, bn_before)
        assert bn!='source_stats' or drift==0
        drifts.append(dict(method=method, bn_buffer_drift=drift, bn_affine_parameter_drift=a.parameter_drift(model, affine_before, trainable_only=False),
                           crossing_count=int(((pred[:,0]>pred[:,1]) | (pred[:,1]>pred[:,2]) | (pred[:,3]>pred[:,4]) | (pred[:,4]>pred[:,5])).sum())))
        np.save(out / f'{method}_validation_predictions.npy', pred)
        print(method, results[method]['B']['validation_score'], flush=True)
        pd.DataFrame(history).to_csv(out / 'pilot_training_history.csv', index=False)
    pd.DataFrame(metrics).to_csv(out / 'pilot_validation_metrics.csv', index=False)
    pd.DataFrame(inventories).to_csv(out / 'pilot_parameter_inventory.csv', index=False)
    pd.DataFrame(drifts).to_csv(out / 'pilot_bn_drift.csv', index=False)
    report = '# Validation-only pilot v2\n\n' + pd.DataFrame(metrics).to_markdown(index=False) + '\n\n' + pd.DataFrame(drifts).to_markdown(index=False) + '\n\nNo test truth was loaded. Readiness awaits independent contract tests.\n'
    (out / 'PILOT_REPORT_VALIDATION_ONLY.md').write_text(report)
    inputs = [SOURCE,SOURCE_GRAPH_CACHE,TARGET_CACHE,FROZEN/'protocol.json',FROZEN/'split_manifest.csv',FROZEN/'filtered_canonical_25g.csv',FROZEN/'filtered_features_25g.csv',Path(__file__),ROOT/'src/qgeognn_al/transfer/adaptation.py',ROOT/'src/qgeognn_al/transfer/source_anchored.py']
    write_json(out / 'pilot_artifact_manifest.json', dict(status='COMPLETED_VALIDATION_ONLY_PENDING_TEST_GATE', **audit,
        results=results, wall_seconds=time.time()-started, inputs={str(p.relative_to(ROOT)):sha(p) for p in inputs},
        artifacts={p.name:sha(p) for p in out.iterdir() if p.is_file()}))

if __name__=='__main__':
    main()
